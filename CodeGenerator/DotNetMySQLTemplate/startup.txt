using System.Text;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.IdentityModel.Tokens;
using Microsoft.AspNetCore.Builder;
using Microsoft.AspNetCore.Hosting;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using {projectName}.Manager.RabitMQAPI.API;
using {projectName}.DataAccess.Impl;
using {projectName}.DataAccess.Interface;
using {projectName}.Manager.Impl;
using {projectName}.Manager.Interface;
using {projectName}.Utility;
using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Routing;
using Microsoft.Extensions.Hosting;
using Microsoft.OpenApi;
using RabbitMQ.Client;
using System.Collections.Generic;
using System;
using System.Linq;
using {projectName}.API.Attributes;
using StackExchange.Redis;

namespace {projectName}.API
{
    public class Startup
    {
        public Startup(IConfiguration configuration)
        {
            Configuration = configuration;
        }

        public IConfiguration Configuration { get; }

        // This method gets called by the runtime. Use this method to add services to the container.
        public void ConfigureServices(IServiceCollection services)
        {
            services.AddSwaggerGen(c =>
                {
                    c.SwaggerDoc("v1", new OpenApiInfo { Title = "{projectName}", Version = "v1" });
                    c.AddSecurityDefinition(name: "Bearer", securityScheme: new OpenApiSecurityScheme
                    {
                        Name = "Authorization",
                        Description = "Enter the Bearer Authorization string as following: `Bearer Generated-JWT-Token`",
                        In = ParameterLocation.Header,
                        Type = SecuritySchemeType.ApiKey,
                        Scheme = "Bearer"
                    });
                    c.AddSecurityRequirement(document => new OpenApiSecurityRequirement
                    {
                        [new OpenApiSecuritySchemeReference("Bearer", document)] = new List<string>()
                    });
                });
             services.AddCors(options =>
            {
                // Cors:AllowedOrigins (comma separated, written by neoapps for the generated UI); any origin only in Development
                var origins = ApiHardening.AllowedOrigins(Configuration);
                options.AddPolicy("AllowAllHeaders",
                   builder =>
                   {
                       builder.AllowAnyHeader().AllowAnyMethod().WithExposedHeaders("X-Correlation-Id");
                       if (origins.Length > 0) builder.WithOrigins(origins);
                       else if (ApiHardening.IsDevelopment) builder.AllowAnyOrigin();
                       else builder.SetIsOriginAllowed(_ => false);
                   });
            });
            services.AddSingleton<IConnectionMultiplexer>(x =>
            ConnectionMultiplexer.Connect(Configuration.GetConnectionString("Redis")));
        services.AddScoped<IRabitMQProducer, RabitMQProducer>();
            services.AddScoped<IRabitMQAsyncProducer, RabitMQAsyncProducer>();
            services.AddControllers();

            var appSettingsSection = Configuration.GetSection("AppSettings");
            services.Configure<AppSettings>(appSettingsSection);

            // configure jwt authentication
            var appSettings = appSettingsSection.Get<AppSettings>();
            var key = Encoding.ASCII.GetBytes(appSettings.Secret);
            services.AddAuthentication(x =>
            {
                x.DefaultAuthenticateScheme = JwtBearerDefaults.AuthenticationScheme;
                x.DefaultChallengeScheme = JwtBearerDefaults.AuthenticationScheme;
            })
            .AddJwtBearer(x =>
            {
                x.RequireHttpsMetadata = false;
                x.SaveToken = true;
                x.TokenValidationParameters = new TokenValidationParameters
                {
                    ValidateIssuerSigningKey = true,
                    IssuerSigningKey = new SymmetricSecurityKey(key),
                    ValidateIssuer = false,
                    ValidateAudience = false,
                    ClockSkew = TimeSpan.FromMinutes(1)
                };
                // revoked sessions (sign-out, password/role change, deactivated account) are refused on every request
                x.Events = new JwtBearerEvents
                {
                    OnTokenValidated = ctx =>
                    {
                        var uid = int.TryParse(ctx.Principal?.FindFirst(System.Security.Claims.ClaimTypes.NameIdentifier)?.Value, out var u) ? u : 0;
                        var tv = int.TryParse(ctx.Principal?.FindFirst(AuthGuard.VersionClaim)?.Value, out var v) ? v : 1;
                        if (!AuthGuard.TokenIsCurrent(uid, tv)) ctx.Fail("This session has ended. Sign in again.");
                        return System.Threading.Tasks.Task.CompletedTask;
                    }
                };
            });
            // sign-in rate limit per client IP (Auth:LoginsPerMinutePerIp)
            var loginsPerMinute = Math.Max(1, Configuration.GetValue("Auth:LoginsPerMinutePerIp", 20));
            services.AddRateLimiter(o =>
            {
                o.RejectionStatusCode = 429;
                o.AddPolicy("login", ctx => System.Threading.RateLimiting.RateLimitPartition.GetFixedWindowLimiter(
                    ctx.Connection.RemoteIpAddress?.ToString() ?? "unknown",
                    _ => new System.Threading.RateLimiting.FixedWindowRateLimiterOptions { PermitLimit = loginsPerMinute, Window = TimeSpan.FromMinutes(1) }));
            });

            services.AddSingleton<IConfiguration>(Configuration);
            string connectionString = Configuration.GetConnectionString("MySQLDatabase");
        string rabbitMqConnection = Configuration.GetConnectionString("RabitMQ");
            ConnectionFactory factory = new ConnectionFactory();
            factory.Uri = new Uri(rabbitMqConnection);
            IConnection connection = factory.CreateConnection();
            services.AddSingleton<IConnection>(connection);
            //services.AddTransient(_ => new MySqlDatabase(connectionString));
            services.AddTransient(_ => new MySqlDatabaseConnector(connectionString));
        services.AddTransient(_ => new RabbitMqConnection(rabbitMqConnection));
            //services.AddSwaggerDocument(c=>c.Title="{projectName}");
            
            services.AddTransient<ICurrentUserService, CurrentUserService>();
            services.AddSingleton<IHttpContextAccessor, HttpContextAccessor>();
            services.AddSingleton<IChannelManager, ChannelManager>();
            // AI assistant (Claude API). Enabled when ANTHROPIC_API_KEY is set.
            services.AddSingleton<{projectName}.API.Ai.AiSettings>();
            services.AddSingleton<{projectName}.API.Ai.AiData>();
            services.AddHttpClient<{projectName}.API.Ai.ClaudeClient>(h => h.Timeout = System.TimeSpan.FromMinutes(5));
            services.AddHttpClient("clinicaltrials");
            services.AddScoped<{projectName}.API.Ai.AiService>();
            #region Dependency
            {serviceDependency}
            #endregion
            
        }

        // This method gets called by the runtime. Use this method to configure the HTTP request pipeline.
        public void Configure(IApplicationBuilder app, IWebHostEnvironment env)
        {
            if (env.IsDevelopment())
            {
                app.UseDeveloperExceptionPage();
            }

            ApiHardening.Configure(Configuration);
            app.Use((ctx, next) => ApiHardening.Middleware(ctx, () => next()));
            if (env.IsDevelopment() || Configuration.GetValue("Swagger:Enabled", false))
            {
                app.UseSwagger();
                app.UseSwaggerUI(c =>
                {
                    c.SwaggerEndpoint("/swagger/v1/swagger.json", "{projectName}");
                });
            }

            // Data rules (tenancy, lifecycle, audit/outbox, read-only data) enforced on every generated query: see Utility/DataGuard*.cs
            {projectName}.Utility.DataGuard.Init(app.ApplicationServices.GetService<IHttpContextAccessor>(),
                Configuration.GetConnectionString("MySQLDatabase"), env.ContentRootPath);
            app.Use(async (ctx, next) =>
            {
                try { await next(); }
                catch ({projectName}.Utility.DataGuardException gx)
                {
                    if (ctx.Response.HasStarted) throw;
                    ctx.Response.StatusCode = gx.StatusCode;
                    await ctx.Response.WriteAsJsonAsync(new { code = 0, message = gx.Message });
                }
            });
            {projectName}.Utility.GuardJobs.Start(app.ApplicationServices);
            AuthGuard.Init(Configuration, Configuration.GetConnectionString("MySQLDatabase"));

            app.UseRouting();
            app.UseRateLimiter();
            app.UseCors("AllowAllHeaders");
            app.UseAuthentication();
            app.UseAuthorization();
            app.UseEndpoints(endpoints =>
            {
                endpoints.MapControllers();
                endpoints.MapGet("/health", () => Results.Ok(new { status = "ok" }));
                // readiness: the API can reach its database, cache and message broker
                endpoints.MapGet("/health/ready", async (IConnectionMultiplexer redis) =>
                {
                    var checks = new Dictionary<string, string>();
                    try { using var c = new MySqlConnector.MySqlConnection(Configuration.GetConnectionString("MySQLDatabase")); await c.OpenAsync(); using var q = new MySqlConnector.MySqlCommand("SELECT 1", c); await q.ExecuteScalarAsync(); checks["mysql"] = "ok"; }
                    catch { checks["mysql"] = "unreachable"; }
                    try { await redis.GetDatabase().PingAsync(); checks["redis"] = "ok"; } catch { checks["redis"] = "unreachable"; }
                    var rabbit = Configuration.GetConnectionString("RabitMQ");
                    if (!string.IsNullOrEmpty(rabbit))
                        try { using var conn = new ConnectionFactory { Uri = new Uri(rabbit), RequestedConnectionTimeout = TimeSpan.FromSeconds(3) }.CreateConnection(); checks["rabbitmq"] = "ok"; }
                        catch { checks["rabbitmq"] = "unreachable"; }
                    var ok = checks.Values.All(v => v == "ok");
                    return Results.Json(new { status = ok ? "ready" : "degraded", checks }, statusCode: ok ? 200 : 503);
                });
            });
        }
    }
}
